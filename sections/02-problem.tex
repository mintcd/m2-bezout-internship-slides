\section{Problem}

\begin{frame}{A Random Model of Regular Expressions}

  Let $\Sigma$ be an alphabet. A BST-like distribution with parameters $u,v \in [0,1]$ selects the root of a random tree of size $n$ as follows:

  \begin{itemize}
    \item If $n=1$, select a letter $a\in \Sigma$ uniformly at random.
    \item If $n=2$, select a Kleene star (\texttt{*}).
    \item If $n \ge 3$, then the root is chosen as
          \begin{itemize}
            \setbeamertemplate{itemize subitem}{\(\circ\)}
            \item concatenation (\texttt{.}) with probability $u$,
            \item alternation (\texttt{|}) with probability $v$,
            \item Kleene star (\texttt{*}) with probability $1-u-v$.
          \end{itemize}
          If a binary operator was chosen, choose $i\in\{1,\ldots,n-1\}$ uniformly at random, then the left and right subtrees are sampled i.i.d. with sizes $i$ and $n-1-i$, respectively.
  \end{itemize}
  \only<7>{Maximum-likelihood fitting on Snort rules \footnote{Cisco's Snort Rules Database \url{https://www.snort.org/}} gives approximately $u \simeq 0.9$ and $v \simeq 0.09$.}

\begin{frame}{The Longest-Literal Recurrence}
  The statistic of interest is
  $$Y_n = \text{length of the longest literal extracted from a random tree of size } n.$$

  Base cases:
  \begin{itemize}
    \item $Y_1=1$: a letter
    \item $Y_2=0$: \texttt{*}-rooted tree is ignored, since its language contains the empty string.
  \end{itemize}

  Let $U_n$ be uniform on $\{1,\ldots,n\}$. Let $I$ and $J$ be indicators of expectations $u$ and $v$ such that $I+J \le 1$. Superscripts are used to denote independent copies.

  $$Y_{n+2} \stackrel{d}{=} I^{(n)}\underbrace{\left(Y_{U_n}^{(1)} + Y_{n+1-U_n}^{(2)}\right)}_{\substack{\text{take \textit{all}}\\ \text{of the subtrees}}} \,\,+\,\, J^{(n)} \underbrace{\max\left(Y_{U_n}^{(1)}, Y_{n+1-U_n}^{(2)}\right)}_{\substack{\text{take \textit{the longer}}\\\text{of the subtrees}}},\quad n\ge 1.$$

  \textbf{Problem:} Analyze the asymptotic behavior of $\EE[Y_n]$. In particular, is it true that $\EE[Y_n] \sim Cn^\alpha$ for some $C, \alpha>0$?
\end{frame}
