\section{Deterministic Lower Bound}

\begin{frame}{Bounding the Expectation}

  $$Y_{n+2} \stackrel{d}{=} I^{(n)}\left(Y_{U_n}^{(1)} + Y_{n+1-U_n}^{(2)}\right) \,\,+\,\, J^{(n)} \max\left(Y_{U_n}^{(1)}, Y_{n+1-U_n}^{(2)}\right),\quad n\ge 1, \quad Y_1=1,Y_2=0.$$

  Take the expectation of both sides. The difficulty is the term
  \[
    \EE\left[\max\left(Y_i^{(1)},Y_{n+1-i}^{(2)}\right)\right].
  \]
  Using
  \[
    \EE[\max(X,Y)]\geq \max(\EE[X],\EE[Y]),
  \]
  we get a deterministic lower-bound recurrence:
  \[
    z_{n+2}
    =
    \dfrac{1}{n}\sum_{i=1}^{n} \left(u\left(z_i+z_{n+1-i}\right)
    +
    v\max(z_i,z_{n+1-i})\right),
    \qquad z_1=1,\;z_2=0.
  \]
\end{frame}

\begin{frame}{Main Deterministic Result}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      A power-law ansatz $z_n\sim Cn^\alpha$ gives
      \[
        \alpha + v2^{-\alpha}=2(u+v)-1.
      \]

      \vspace{0.2cm}
      \begin{mytheorem}[Exact asymptotics]
        If $u>1/2$, then
        \[
          z_n\sim Cn^\alpha
        \]
        for some $C>0$, where $\alpha$ solves the equation above.
      \end{mytheorem}
    \end{column}

    \begin{column}{0.43\textwidth}
      \includegraphics[width=\linewidth]{ratio-strict-lower-bound.png}
    \end{column}
  \end{columns}

  \vspace{0.2cm}

  We take the ansatz $z_n\sim Cn^\alpha$ to find out intermediate results that must be true.
\end{frame}
